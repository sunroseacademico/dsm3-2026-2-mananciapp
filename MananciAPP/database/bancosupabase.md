-- ============================================================
-- BANCO IGREJA / CURSOS - SUPABASE (POSTGRES)
-- Baseado no script que você confirmou que RODA nesse projeto (com aspas
-- nos identificadores) — só que aqui todo identificador está em minúsculo
-- DENTRO das aspas. Motivo: o app inteiro escreve SQL sem aspas
-- (ex.: "FROM ID_FIEL"), e sem aspas o Postgres sempre dobra o nome pra
-- minúsculo antes de comparar. Então "id_fiel" (com aspas, minúsculo) é
-- o único jeito de bater com o que o app manda, SEM abrir mão das aspas
-- que o seu editor precisa pra aceitar o script.
--
-- Outras diferenças em relação ao script original:
-- 1. INTEGER no lugar de BOOLEAN nas colunas de flag — o código injeta
--    literais 0/1 direto no SQL; Postgres não faz cast implícito de
--    integer pra boolean.
-- 2. Colunas/tabelas que faltavam (usadas no código via tabela_tem_coluna()/
--    site_tabela_existe(), mas ausentes no script original): idc_professor_nome,
--    idc_nivel, idc_destaque (curso); idmatr_inicio_em (matrícula);
--    idpr_tempo_assistido_segundos, idpr_ultimo_acesso_em (progresso);
--    idf_last_activity (fiel); tabelas id_favorito e id_historico_visualizacao.
-- 3. UNIQUE (idmatr_id, idct_id) em id_progresso e UNIQUE (idf_id, idc_id)
--    em id_favorito — necessárias pros upserts (ON CONFLICT) do código.
--
-- Sem dados reais a preservar (confirmado com o usuário) — schema criado do
-- zero.
-- ============================================================

-- ============================================================
-- 12-B. TENTATIVAS DE LOGIN (anti força-bruta)
-- ============================================================
-- Sem isso, login.php não tinha NENHUMA proteção contra tentativa repetida
-- de senha (nem delay, nem captcha, nem bloqueio) — um script podia tentar
-- milhares de senhas por minuto contra qualquer e-mail conhecido. Cada
-- tentativa de senha ERRADA grava uma linha aqui; se acumular
-- LOGIN_MAX_TENTATIVAS (ver login.php) dentro da janela de tempo, o login
-- fica bloqueado pra aquele e-mail até as tentativas mais antigas saírem da
-- janela. Um login bem-sucedido apaga as tentativas daquele e-mail.

CREATE TABLE "id_login_tentativa" (
    "idlt_id" BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    "idlt_email" VARCHAR(100) NOT NULL,
    "idlt_criado_em" TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP

-- ============================================================
-- 25. USUÁRIOS REAIS (acessos já usados antes da migração)
-- ============================================================
-- Mesmos 4 acessos que já existiam no schema MySQL antigo (ver
-- database/Banco Igreja.sql e database/Banco Igreja Robusto.sql) — trazidos
-- pra cá em modo "robusto" (IDF_SENHA_HASH com bcrypt de verdade via
-- password_hash(), não IDF_SENHA em texto puro), já que é o modo padrão do
-- app hoje (MODO_BANCO em php/conexao.php).
--
-- Sem senha: os 4 entram com idf_senha_hash = '!definir-senha', que não é
-- um hash válido — ninguém consegue entrar até alguém definir a senha. (Até
-- out/2026 todos vinham com a mesma senha fraca, escrita aqui e no README;
-- quem lesse o repositório entrava como admin em qualquer ambiente criado
-- com este arquivo.) Logo depois de rodar, defina a senha do admin aqui no
-- SQL Editor, trocando SUA_SENHA (o app aceita esse hash e o regrava no
-- formato dele no primeiro login):
--
--   UPDATE "id_fiel"
--   SET "idf_senha_hash" = extensions.crypt('SUA_SENHA', extensions.gen_salt('bf', 10))
--   WHERE "idf_email" = 'admin@igreja.com';
--
-- Os fiéis de exemplo usam "Esqueci minha senha" no login, ou o admin manda
-- o link pelo botão da chave em Gerenciar Usuários.
--
-- E-mails: admin@igreja.com (administrador), joao@email.com e
-- maria@email.com (Aluno), carlos@email.com (Líder).
--

-- ============================================================
-- 29. CORRIGE "idmatr_percentual" TRAVADO EM 0% AO CONCLUIR UM CURSO
-- ============================================================
-- Bug: curso_calcular_progresso() (php/fiel/includes/cursos_helpers.php) faz
-- UPDATE "id_matricula" SET ..., "idmatr_conclusao_em" = NOW(), ...  sempre
-- que o percentual chega a 100% — mas essa coluna nunca existiu neste schema
-- Postgres (só no schema MySQL antigo). O UPDATE falhava por completo (erro
-- silencioso, o driver PDO deste app não propaga o erro pra tela), então
-- IDMATR_PERCENTUAL nunca chegava a ser gravado como 100 — ficava travado no
-- valor default (0) mesmo com todas as aulas concluídas em "id_progresso".
-- Percentuais parciais (&lt;100%) não eram afetados, só a conclusão total.
--
-- 1) Adiciona a coluna que faltava (idempotente — não falha se já existir).
-- 2) Recalcula "idmatr_percentual"/"idmatr_status" de TODAS as matrículas a
--    partir de "id_progresso" (fonte da verdade), corrigindo quem ficou
--    travado em 0% por causa desse bug. Seguro rodar de novo quando quiser.

ALTER TABLE "id_matricula" ADD COLUMN IF NOT EXISTS "idmatr_conclusao_em" TIMESTAMPTZ;

UPDATE "id_matricula" m
SET "idmatr_percentual" = sub.pct,
    "idmatr_status" = CASE WHEN sub.pct >= 100 THEN 'CONCLUIDA' ELSE m."idmatr_status" END,
    "idmatr_conclusao_em" = CASE WHEN sub.pct >= 100 THEN COALESCE(m."idmatr_conclusao_em", CURRENT_TIMESTAMP) ELSE m."idmatr_conclusao_em" END
FROM (
    SELECT mt."idmatr_id",
           ROUND(100.0 * COALESCE(cc.concluidas, 0) / ct.total, 2) AS pct
    FROM "id_matricula" mt
    JOIN (
        SELECT "idc_id", COUNT(*) AS total
        FROM "id_content"
        WHERE LOWER("idct_tipo") = 'video' AND "idct_publicado" = 1
        GROUP BY "idc_id"
    ) ct ON ct."idc_id" = mt."idc_id"
    LEFT JOIN (
        SELECT "idmatr_id", COUNT(*) AS concluidas
        FROM "id_progresso"
        WHERE "idpr_status" = 'CONCLUIDO'
        GROUP BY "idmatr_id"
    ) cc ON cc."idmatr_id" = mt."idmatr_id"
) sub
WHERE m."idmatr_id" = sub."idmatr_id";

-- 30. CONTROLE DE PUSH JÁ ENVIADO POR NOTIFICAÇÃO
-- ============================================================
-- OneSignal.login() (vincula o navegador ao IDF_ID) só roda no carregamento
-- da página SEGUINTE ao login — no exato momento em que a notificação do
-- certificado de boas-vindas é criada (durante o próprio POST de login),
-- ainda não dá tempo desse vínculo existir. Por isso o envio do push não
-- acontece mais junto da criação da notificação: fica pendente até o
-- navegador confirmar (via AJAX, depois do OneSignal.login() resolver no
-- JS) que já está pronto — ver notificacoes_push_enviar_pendentes() e
-- php/fiel/ajax_onesignal_sync.php. "idn_push_enviado" marca o que já foi
-- efetivamente entregue à OneSignal, pra não reenviar a mesma notificação
-- toda vez que essa checagem rodar.
--
-- Ao adicionar a coluna, TODAS as notificações que já existem hoje (ex.:
-- "Acesso Aprovado!" de quem já foi aprovado há semanas) precisam ficar
-- marcadas como IDN_PUSH_ENVIADO = 1 (não 0) — senão, no próximo login de
-- cada um desses fiéis, o sistema tentaria mandar push retroativo de coisa
-- que já aconteceu há muito tempo. Só notificações CRIADAS DAQUI PRA FRENTE
-- devem nascer com IDN_PUSH_ENVIADO = 0 (valor default da coluna).
--
-- Exceção: notificações de certificado de boas-vindas ("certificado_boas_
-- vindas") que já existirem ficam de fora desse backfill — continuam
-- pendentes, pra você já poder ver o push funcionando com quem já ganhou o
-- certificado, sem precisar apagar a notificação e logar de novo.

ALTER TABLE "id_notificacao" ADD COLUMN IF NOT EXISTS "idn_push_enviado" INTEGER NOT NULL DEFAULT 0;
UPDATE "id_notificacao" SET "idn_push_enviado" = 1 WHERE "idn_push_enviado" = 0 AND "idn_tipo" != 'certificado_boas_vindas';

-- 33. REVOGAÇÃO DE SESSÃO DE VERDADE + PARA DE DUPLICAR LINHA EM ID_SESSAO
-- ============================================================
-- "id_sessao" existia só como log (nunca era reconsultada) — auth_logout()
-- gravava IDS_REVOGADA=1 mas nada verificava isso de volta. Agora
-- auth_require() confere isso a cada página, e o admin ganhou um botão de
-- "Desconectar" (força logout de sessão ativa de outro usuário).

-- Efeito colateral do fix anterior (SESSION_TIMEOUT_SECONDS): auth_touch()
-- passou a rodar a cada página, e auth_store_session() sempre fez um INSERT
-- puro (sem chave única) — ou seja, desde aquele deploy, toda página vista
-- por um usuário logado criava uma linha NOVA em "id_sessao" (antes só
-- acontecia 1x, no login). Sem worries de dado incorreto, mas a tabela
-- estava crescendo rápido demais. Isso:
-- 1) Apaga as duplicatas por token, mantendo só a mais recente de cada.
-- 2) Adiciona a UNIQUE que faltava em "ids_token" (só dá pra adicionar
--    depois de remover as duplicatas).
-- auth_store_session() (php/conexao.php) passou a fazer UPSERT (ON
-- CONFLICT), então não duplica mais daqui pra frente.

DELETE FROM "id_sessao" a
USING "id_sessao" b
WHERE a."ids_token" = b."ids_token"
  AND a."ids_token" IS NOT NULL
  AND a."ids_id" < b."ids_id";

-- A ordem do CASE importa: "Pastor de jovens" vira Pastor(a), "Líder de

-- 66. E-MAIL ÚNICO SEM DIFERENCIAR MAIÚSCULAS
-- ============================================================
-- O UNIQUE de idf_email diferencia maiúsculas: "Fulano@x.com" e

-- e o login abria a outra. Este índice impede de vez (o cadastro já trata a
-- violação com "Este e-mail já está cadastrado"; a edição pelo admin também
-- confere antes).
-- ANTES de criar o índice, rode a consulta abaixo. Se ela devolver alguma
-- linha, há contas duplicadas: decida com o cliente qual fica (ou troque o
-- e-mail de uma delas no painel) e só depois crie o índice — senão ele falha
-- com "could not create unique index".
--
--   SELECT LOWER(idf_email) AS email, array_agg(idf_id ORDER BY idf_id) AS contas
--   FROM id_fiel
--   GROUP BY LOWER(idf_email)
--   HAVING COUNT(*) > 1;
--
-- Idempotente: pode rodar de novo sem erro.

CREATE UNIQUE INDEX IF NOT EXISTS "ux_id_fiel_email_lower" ON "id_fiel" (LOWER("idf_email"));


-- ============================================================
-- 67. FIM DAS SENHAS EM TEXTO PURO
-- ============================================================
-- Contas antigas (de antes do hash) podem ter a senha legível em idf_senha.
-- O login já troca por hash quando a pessoa entra, mas quem nunca mais
-- entrou continuava com a senha exposta. Aqui o próprio Postgres gera o hash
-- bcrypt (pgcrypto, já instalado no Supabase no schema "extensions") e apaga
-- o texto puro de TODAS as contas de uma vez. O PHP aceita esse hash
-- ("$2a$...") no password_verify, e o login o regrava no formato atual
-- ("$2y$") na primeira entrada (password_needs_rehash, login.php). Ninguém
-- precisa trocar de senha.
-- Para ver o tamanho do problema antes:
--
--   SELECT COUNT(*) FILTER (WHERE COALESCE(idf_senha, '') <> '') AS com_texto_puro,
--          COUNT(*) FILTER (WHERE COALESCE(idf_senha, '') <> '' AND COALESCE(idf_senha_hash, '') = '') AS so_texto_puro
--   FROM id_fiel;
--
-- Contas que têm os dois (hash e texto puro): o login usa só o hash, então o
-- texto puro é sobra e só é apagado.
-- Idempotente: pode rodar de novo sem erro (na segunda vez não muda nada).

CREATE EXTENSION IF NOT EXISTS pgcrypto WITH SCHEMA extensions;

UPDATE "id_fiel"
SET "idf_senha_hash" = extensions.crypt("idf_senha", extensions.gen_salt('bf', 10))
WHERE COALESCE("idf_senha_hash", '') = '' AND COALESCE("idf_senha", '') <> '';

UPDATE "id_fiel"
SET "idf_senha" = NULL
WHERE "idf_senha" IS NOT NULL;

-- 69. CERTIFICADOS DE CURSOS JÁ CONCLUÍDOS
-- ============================================================
-- Até out/2026 concluir um curso não emitia certificado (só existia o de
-- boas-vindas). O app agora emite na hora em que a matrícula chega a 100%
-- (certificado_curso_conceder() em php/conexao.php). Este SQL emite os

-- assistir de novo a alguma aula do curso.
-- Idempotente: pode rodar de novo sem erro (não duplica).

INSERT INTO "id_certificado" ("idf_id", "idc_id", "idcert_numero", "idcert_data_conclusao")
SELECT m."idf_id", m."idc_id", 'CC-' || m."idc_id" || '-' || m."idf_id", COALESCE(m."idmatr_conclusao_em", NOW())
FROM "id_matricula" m
JOIN "id_curso" c ON c."idc_id" = m."idc_id" AND c."idc_titulo" <> 'Boas-vindas'
WHERE m."idmatr_status" = 'CONCLUIDA'
  AND NOT EXISTS (SELECT 1 FROM "id_certificado" x WHERE x."idf_id" = m."idf_id" AND x."idc_id" = m."idc_id")
ON CONFLICT ("idcert_numero") DO NOTHING;
